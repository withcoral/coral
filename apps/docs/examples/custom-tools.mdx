Coral saves the tool-calling code that an agent writes as a custom tool.
The next agent calls that tool and does not write the code again.
This page shows the effect on real tasks.

The agent ran each task twice:

- **Without custom tools.** The agent started cold, with only the provider tools (Slack, Linear, GitHub).
- **With custom tools.** The agent could also call the custom tools that Coral saved from earlier runs.

Across the 34 tasks below, custom tools gave these results:

## Example: find the pull request that fixed a release alarm

A summary of the prompt:

> Someone reported a failed release workflow after a change to its push steps. Which pull request fixed it?

| Agent run | Total tokens | Token cost | Task time |
|---|--:|--:|--:|
| Without custom tools | 423,710 | &#36;0.36 | 78 s |
| With custom tools | 173,032 | &#36;0.13 | 34 s |

The agent used two custom tools.
Each tool is a TypeScript program that calls the provider tools.

```ts searchMessages.ts expandable
export default async function searchMessages(input: { query: string; channels?: string[]; oldest?: string; latest?: string; limit?: number; allChannels?: boolean }): Promise<{
  ok: boolean; error?: string; window: { oldest: string; latest: string };
  channelsScanned: string[]; channelsNotScanned: string[]; skipped: { channel: string; error: string }[];
  matched: number; matches: { channel: string; channelId: string; ts: string; time: string; author: string; thread_ts?: string; replies: number; text: string }[]; note: string;
}> {
  const A: any = (input as any)?.input ?? input ?? {};
  const msg = (e: unknown) => (e instanceof Error ? e.message : String(e));
  const unix = (s?: unknown) => s == null || s === "" ? undefined : /^\d+(\.\d+)?$/.test(String(s)) ? Number(s) : Math.floor(Date.parse(String(s)) / 1000);
  const listChannels = async () => {
    const out: any[] = []; let cursor: string | undefined;
    for (let i = 0; i < 10; i++) {
      const r: any = await coral.providers.slack.conversationsList({ limit: 1000, types: "public_channel,private_channel", cursor });
      if (!r?.data?.ok) throw new Error(`Slack conversations.list failed: ${r?.data?.error ?? r?.metadata?.status}`);
      out.push(...(r.data.channels ?? [])); cursor = r.data.response_metadata?.next_cursor || undefined; if (!cursor) break;
    }
    return out;
  };
  const nameOf = async (ids: string[]) => {
    const names: Record<string, string> = {};
    await Promise.all(Array.from(new Set(ids.filter(Boolean))).slice(0, 20).map(async id => {
      const u: any = await coral.providers.slack.usersInfo({ user: id });
      if (u?.data?.ok) names[id] = u.data.user?.real_name || u.data.user?.profile?.display_name || u.data.user?.name || id;
    }));
    return names;
  };

  const started = Date.now(), now = Math.floor(started / 1000);
  const oldest = unix(A.oldest ?? A.after ?? A.since) ?? now - 120 * 86400, latest = unix(A.latest ?? A.before ?? A.until);
  const window = { oldest: new Date(oldest * 1000).toISOString(), latest: latest ? new Date(latest * 1000).toISOString() : "now" };
  const query = String(A.query ?? A.q ?? A.text ?? "").trim();
  const words = query.toLowerCase().split(/[^a-z0-9#@._-]+/).filter((w: string) => w.length > 1);
  const need = words.length <= 2 ? words.length : Math.ceil(words.length * 0.6);
  const base = { window, channelsScanned: [] as string[], channelsNotScanned: [] as string[], skipped: [] as { channel: string; error: string }[], matched: 0, matches: [] as any[] };
  if (!words.length) return { ok: false, error: "Pass query: the words to look for.", ...base, note: "" };
  let chans: any[];
  try { chans = await listChannels(); } catch (e) { return { ok: false, error: msg(e), ...base, note: "" }; }
  const wanted = (A.channels ?? A.channel ? [].concat(A.channels ?? A.channel) : []).map((c: any) => String(c).replace(/^#/, "").toLowerCase());
  const usable = chans.filter(c => c.is_member && !(latest && c.created > latest) && !(c.is_archived && c.updated && c.updated / 1000 < oldest)
    && (!wanted.length || wanted.includes(String(c.name).toLowerCase()) || wanted.includes(String(c.id).toLowerCase())));
  const score = (c: any) => { const n = String(c.name).toLowerCase(), about = `${c.topic?.value ?? ""} ${c.purpose?.value ?? ""}`.toLowerCase();
    return words.reduce((s: number, w: string) => s + (n.includes(w) ? 3 : 0) + (about.includes(w) ? 1 : 0), 0); };
  const ranked = usable.map(c => ({ c, s: score(c) })).sort((a, b) => b.s - a.s || (b.c.updated ?? 0) - (a.c.updated ?? 0)).map(x => x.c);
  const cap = A.allChannels || wanted.length ? ranked.length : Math.min(ranked.length, Number(A.maxChannels ?? 15));
  const limit = Math.min(Number(A.limit ?? 15), 40);
  const scanned: string[] = [], skipped: { channel: string; error: string }[] = [], hits: any[] = [];
  const scan = async (c: any) => {
    let cursor: string | undefined;
    for (let p = 0; p < 3; p++) {
      const r: any = await coral.providers.slack.conversationsHistory({ channel: c.id, oldest, latest, limit: 200, cursor });
      if (!r?.data?.ok) { skipped.push({ channel: c.name, error: String(r?.data?.error ?? r?.metadata?.status) }); return; }
      for (const m of r.data.messages ?? []) {
        const text = String(m.text ?? ""), low = text.toLowerCase();
        if (words.filter((w: string) => low.includes(w)).length >= need)
          hits.push({ channel: c.name, channelId: c.id, ts: m.ts, time: new Date(Number(m.ts) * 1000).toISOString(), user: m.user, author: m.username ?? m.user ?? "",
            thread_ts: m.thread_ts, replies: m.reply_count ?? 0, text: text.length > 240 ? text.slice(0, 240) + "…" : text });
      }
      cursor = r.data.response_metadata?.next_cursor || undefined; if (!r.data.has_more || !cursor) break;
    }
    scanned.push(c.name);
  };
  let i = 0;
  while (i < cap && Date.now() - started < 20000) {
    await Promise.all(ranked.slice(i, Math.min(i + 3, cap)).map(scan)); i += 3;
    if (!A.allChannels && !wanted.length && hits.length >= limit * 2 && i >= 6) break;
  }
  hits.sort((a, b) => Number(b.ts) - Number(a.ts));
  const shown = hits.slice(0, limit);
  const names = await nameOf(shown.map(m => m.user));
  const matches = shown.map(({ user, ...m }) => ({ ...m, author: names[user] ?? m.author }));
  const notScanned = ranked.slice(Math.min(i, ranked.length)).map(c => c.name);
  const limited = skipped.filter(s => s.error === "ratelimited").map(s => s.channel);
  const note = [notScanned.length ? `${notScanned.length} member channels not scanned (lowest-ranked or out of time); pass channels: [...] or allChannels: true to scan them.` : "",
    limited.length ? `Slack rate-limited ${limited.length} channels; retry with channels: ${JSON.stringify(limited.slice(0, 10))}.` : "",
    "Matches are top-level messages and thread roots; read replies with slack.threadReplies."].filter(Boolean).join(" ");
  return { ok: true, ...base, channelsScanned: scanned, channelsNotScanned: notScanned.slice(0, 40), skipped, matched: hits.length, matches, note };
}
```

### `slack.threadLinkedWork`

Reads a Slack thread and looks up the GitHub pull requests, GitHub issues and Linear issues that it links to.

<CodeGroup>

```ts threadLinkedWork.d.ts
export default function threadLinkedWork(input: {
    channel: string;
    ts: string;
}): Promise<{
    ok: boolean;
    error?: string;
    channel: string;
    messages: {
        time: string;
        author: string;
        text: string;
    }[];
    linked: {
        source: string;
        ref: string;
        title?: string;
        state?: string;
        owner?: string;
        url?: string;
        error?: string;
    }[];
}>;
```

```ts threadLinkedWork.ts expandable
export default async function threadLinkedWork(input: { channel: string; ts: string }): Promise<{
  ok: boolean; error?: string; channel: string; messages: { time: string; author: string; text: string }[];
  linked: { source: string; ref: string; title?: string; state?: string; owner?: string; url?: string; error?: string }[];
}> {
  const A: any = (input as any)?.input ?? input ?? {};
  const msg = (e: unknown) => (e instanceof Error ? e.message : String(e));
  const unix = (s?: unknown) => s == null || s === "" ? undefined : /^\d+(\.\d+)?$/.test(String(s)) ? Number(s) : Math.floor(Date.parse(String(s)) / 1000);
  const listChannels = async () => {
    const out: any[] = []; let cursor: string | undefined;
    for (let i = 0; i < 10; i++) {
      const r: any = await coral.providers.slack.conversationsList({ limit: 1000, types: "public_channel,private_channel", cursor });
      if (!r?.data?.ok) throw new Error(`Slack conversations.list failed: ${r?.data?.error ?? r?.metadata?.status}`);
      out.push(...(r.data.channels ?? [])); cursor = r.data.response_metadata?.next_cursor || undefined; if (!cursor) break;
    }
    return out;
  };
  const nameOf = async (ids: string[]) => {
    const names: Record<string, string> = {};
    await Promise.all(Array.from(new Set(ids.filter(Boolean))).slice(0, 20).map(async id => {
      const u: any = await coral.providers.slack.usersInfo({ user: id });
      if (u?.data?.ok) names[id] = u.data.user?.real_name || u.data.user?.profile?.display_name || u.data.user?.name || id;
    }));
    return names;
  };

  let channel = String(A.channel ?? A.channelId ?? "").replace(/^#/, "");
  const ts = String(A.ts ?? A.thread_ts ?? "");
  if (!channel || !ts) return { ok: false, error: "Pass channel (name or id) and ts (the thread's root timestamp).", channel, messages: [], linked: [] };
  if (!/^[CGD][A-Z0-9]+$/.test(channel)) { try { channel = (await listChannels()).find((c: any) => c.name === channel)?.id ?? channel; } catch (e) { return { ok: false, error: msg(e), channel, messages: [], linked: [] }; } }
  const r: any = await coral.providers.slack.conversationsReplies({ channel, ts: Number(ts), limit: 200 });
  if (!r?.data?.ok) return { ok: false, error: `Slack conversations.replies failed: ${r?.data?.error ?? r?.metadata?.status}`, channel, messages: [], linked: [] };
  const raw: any[] = r.data.messages ?? [];
  const names = await nameOf(raw.map(m => m.user));
  const messages = raw.map(m => ({ time: new Date(Number(m.ts) * 1000).toISOString(), author: names[m.user] ?? m.user ?? "", text: String(m.text ?? "").slice(0, 1200) }));
  const all = raw.map(m => String(m.text ?? "")).join("\n");
  const linked: any[] = []; const seen = new Set<string>();
  for (const m of all.matchAll(/https:\/\/github\.com\/([\w.-]+)\/([\w.-]+)\/(pull|issues)\/(\d+)/g)) {
    if (seen.has(m[0]) || linked.length >= 15) continue; seen.add(m[0]);
    const g: any = m[3] === "pull" ? await coral.providers.github.pulls._get({ owner: m[1], repo: m[2], pull_number: Number(m[4]) }) : await coral.providers.github.issues._get({ owner: m[1], repo: m[2], issue_number: Number(m[4]) });
    linked.push(g?.metadata?.status < 300 ? { source: "github", ref: `${m[2]}#${m[4]}`, title: g.data.title, state: g.data.merged_at ? "merged" : g.data.state, owner: g.data.user?.login, url: m[0] } : { source: "github", ref: m[0], error: `status ${g?.metadata?.status}` });
  }
  const parse = (x: any) => { try { return JSON.parse((x?.data?.content ?? []).map((c: any) => c?.text ?? "").join("") || "{}"); } catch { return {}; } };
  for (const m of all.matchAll(/\b([A-Z][A-Z0-9]{1,9}-\d+)\b/g)) {
    if (seen.has(m[1]) || linked.length >= 15) continue; seen.add(m[1]);
    const d = parse(await coral.providers.linear.getIssue({ id: m[1], fields: ["id", "title", "status", "assignee", "team", "url"] }));
    if (d?.id) linked.push({ source: "linear", ref: d.id, title: d.title, state: d.status, owner: d.assignee, url: d.url });
  }
  return { ok: true, channel, messages, linked };
}
```
